async function exportPDF(elId,filename){

 const{jsPDF}=window.jspdf;

 const el=$(elId);

 if(!el){showToast("لا يوجد تقرير للمعاينة");return;}

 

 showToast("جاري تجهيز معاينة الـ PDF...");

 const originalWidth=el.style.width;

 const originalOverflow=el.style.overflow;

 el.style.width="820px";

 el.style.overflow="visible";

 

 try{

  const canvas=await html2canvas(el,{scale:2.5,backgroundColor:"#ffffff",windowWidth:820});

  const imgData=canvas.toDataURL("image/jpeg",0.98);

  const pdf=new jsPDF("p","mm","a4",true);

  const pw=pdf.internal.pageSize.getWidth();

  const ph=pdf.internal.pageSize.getHeight();

  const imgW=pw;

  const imgH=canvas.height*pw/canvas.width;

  let heightLeft=imgH,position=0;

 

  pdf.addImage(imgData,"JPEG",0,position,imgW,imgH);

  heightLeft-=ph;

  while(heightLeft>0){

   position=heightLeft-imgH;

   pdf.addPage();

   pdf.addImage(imgData,"JPEG",0,position,imgW,imgH);

   heightLeft-=ph;

  }

 

  // إنتاج رابط معاينة Blob وعرضه داخل إطار داخلي بدلاً من فتح شاشة الطباعة

  const pdfBlob=pdf.output("blob");

  if(currentPdfBlobUrl) URL.revokeObjectURL(currentPdfBlobUrl);

  currentPdfBlobUrl=URL.createObjectURL(pdfBlob);

 

  $("pdf-preview-iframe").src=currentPdfBlobUrl;

  $("download-pdf-btn").onclick=()=>{

   pdf.save(`${filename}_${todayISO()}.pdf`);

   showToast("تم تنزيل ملف الـ PDF ✅");

  };

 

  $("pdf-preview-modal").classList.remove("hidden");

function dayBefore(iso){const d=new Date(iso+"T12:00:00");d.setDate(d.getDate()-1);return`${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,"0")}-${String(d.getDate()).padStart(2,"0")}`}

function shiftISO(iso,days){const d=new Date(iso+"T12:00:00");d.setDate(d.getDate()+days);return`${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,"0")}-${String(d.getDate()).padStart(2,"0")}`}

function cyclePeriods(records){

 const today=todayISO(),bases=records.filter(r=>r.type==="income"&&r.isSalaryBase&&r.date<=today).map(r=>r.date).sort();

 const unique=[...new Set(bases)],start=unique.length?unique[unique.length-1]:firstOfMonthISO();

 let previousStart,previousEnd;

 if(unique.length>=2){

  previousStart=unique[unique.length-2];

  previousEnd=dayBefore(start);

 }else{

  const days=Math.max(1,Math.round((new Date(today+"T12:00:00")-new Date(start+"T12:00:00"))/86400000)+1);

  previousEnd=dayBefore(start);

  previousStart=shiftISO(start,-days);

 }

 return{start,end:today,previousStart,previousEnd,hasSalary:unique.length>0,previousHasSalary:unique.length>=2}

}

function getRange(prefix){return{from:readDateParts(prefix+"-from"),to:readDateParts(prefix+"-to")}}

function inPeriod(r,p){return r.date>=p.from&&r.date<=p.to}

function sumWhere(records,fn){return records.reduce((s,r)=>s+(fn(r)?realAmount(r):0),0)}

function incomeFor(records,p){return sumWhere(records,r=>inPeriod(r,p)&&r.type==="income")}

function expenseFor(records,p){return sumWhere(records,r=>inPeriod(r,p)&&r.type==="expense")}

function transferFor(records,p){return sumWhere(records,r=>inPeriod(r,p)&&r.type==="transfer")}

function debtTakenFor(records,p){return sumWhere(records,r=>inPeriod(r,p)&&isCreditorEntry(r))}

function debtPaidFor(records,p){return sumWhere(records,r=>inPeriod(r,p)&&isDebtorEntry(r))}

function groupAmounts(records,p,filter,keyFn){

 const out={};

 records.filter(r=>inPeriod(r,p)&&filter(r)).forEach(r=>{

  const k=keyFn(r)||"غير مصنف";

  out[k]=(out[k]||0)+realAmount(r);

 });

 return Object.entries(out).sort((a,b)=>b[1]-a[1]);

}

function calculateCarryOver(records,beforeDate){

 const opening=Number(localStorage.getItem(OPENING_BAL_KEY)||0);let balance=opening;

 records.filter(r=>r.date<beforeDate).forEach(r=>{

  if(r.type==="income")balance+=realAmount(r);

  else if(r.type==="expense")balance-=realAmount(r);

  else if(r.type==="transfer")balance-=realAmount(r);

  else if(r.type==="debt"){

   if(r.debtType==="creditor")balance+=realAmount(r);

   else if(r.debtType==="debtor")balance-=realAmount(r);

  }

 });

 return balance;

}

 records.filter(r=>r.date<=today).forEach(r=>{

  if(r.type==="income")balance+=realAmount(r);

function calculateSpentInCycle(records,cycle){

 const period={from:cycle.start,to:cycle.end};

 const result={transfers:0,debt:0,categories:{},extraIncome:0};

 BUDGET_CATEGORY_ITEMS.forEach(c=>result.categories[c.key]=0);

 records.filter(r=>inPeriod(r,period)).forEach(r=>{

  if(r.type==="expense"){

   const cat=r.category||"أخرى";

   result.categories[cat]=(result.categories[cat]||0)+realAmount(r);

  }else if(r.type==="transfer"){

   result.transfers+=realAmount(r);

  }else if(r.type==="debt"){

   if(r.debtType==="debtor")result.debt+=realAmount(r);

  }else if(r.type==="income"&&!r.isSalaryBase){

   result.extraIncome+=realAmount(r);

  }

 });

 return result;

}

function numberToArabicWords(num){

 if(num===0)return "صفر";

 const ones=["","واحد","اثنان","ثلاثة","أربعة","خمسة","ستة","سبعة","ثمانية","تسعة","عشرة",

  "أحد عشر","اثنا عشر","ثلاثة عشر","أربعة عشر","خمسة عشر","ستة عشر","سبعة عشر","ثمانية عشر","تسعة عشر"];

 const tens=["","","عشرون","ثلاثون","أربعون","خمسون","ستون","سبعون","ثمانون","تسعون"];

 const hundreds=["","مائة","مائتان","ثلاثمائة","أربعمائة","خمسمائة","ستمائة","سبعمائة","ثمانمائة","تسعمائة"];

 function under1000(n){

  if(n===0)return "";

  const h=Math.floor(n/100),r=n%100;

  let s=hundreds[h]||"";

  if(r>0){

   if(s)s+=" و";

   if(r<20)s+=ones[r];

   else{

    const t=Math.floor(r/10),o=r%10;

    if(o>0)s+=ones[o]+" و";

    s+=tens[t];

   }

  }

  return s;

 }

 const intPart=Math.floor(num);

 const decPart=Math.round((num-intPart)*1000);

 const thousands=Math.floor(intPart/1000);

 const rest=intPart%1000;

 let words="";

 if(thousands>0){

  if(thousands===1)words="ألف";

  else if(thousands===2)words="ألفان";

  else if(thousands<=10)words=under1000(thousands)+" آلاف";

  else words=under1000(thousands)+" ألف";

  if(rest>0)words+=" و"+under1000(rest);

 }else{

  words=under1000(rest);

 }

 if(!words)words="صفر";

 words+=" دينار كويتي";

 if(decPart>0)words+=" و"+under1000(decPart)+" فلس";

 return words+" لا غير";

}

function buildStatementHTML(person,txs,period){

 const personTxs=txs.filter(t=>t.type==="debt"&&t.person===person)

   .filter(t=>!period||(t.date>=period.from&&t.date<=period.to));

 if(!personTxs.length){

  return `<div class="stmt"><div class="stmt-empty">لا توجد حركات لهذا الطرف.</div></div>`;

 }

 const opening=personTxs.filter(t=>t.debtType==="opening_balance");

 const others=personTxs.filter(t=>t.debtType!=="opening_balance")

   .sort((a,b)=>a.date.localeCompare(b.date)||(a.time||"").localeCompare(b.time||""));

 const rows=[];

 let balance=0,seq=0;

 opening.forEach(r=>{

  const amt=realAmount(r);

  balance+=amt;

  rows.push({seq:++seq,ref:"OPN-"+String(seq).padStart(3,"0"),date:"—",desc:"رصيد افتتاحي",madden:amt,daen:0,balance,opening:true});

 });

 others.forEach(r=>{

  const amt=realAmount(r);

  const isMadden=r.debtType==="creditor";

  const madden=isMadden?amt:0;

  const daen=isMadden?0:amt;

  balance+=madden-daen;

  rows.push({

   seq:++seq,ref:"TX-"+String(seq).padStart(5,"0"),date:fmtDate(r.date),

   desc:r.debtType==="creditor"?"دين جديد (استلام)":"سداد (دفع)",

   madden,daen,balance,opening:false,notes:r.notes||""

  });

 });

 const totalMadden=rows.reduce((s,r)=>s+r.madden,0);

 const totalDaen=rows.reduce((s,r)=>s+r.daen,0);

 const finalBalance=balance;

 const isNegative=finalBalance<0;

 const isZero=finalBalance===0;

 const now=new Date();

 const stamp=`${now.getFullYear()}-${String(now.getMonth()+1).padStart(2,"0")}-${String(now.getDate()).padStart(2,"0")}`;

 const timeStr=`${String(now.getHours()).padStart(2,"0")}:${String(now.getMinutes()).padStart(2,"0")}`;

 const periodText=period

   ?`من ${fmtDate(period.from)} إلى ${fmtDate(period.to)}`

   :`كافة الحركات المسجّلة`;

 let tableRows="";

 rows.forEach(r=>{

  const descHTML=`${escapeHTML(r.desc)}`+

   (r.ref?`<br><small style="color:#94a3b8;font-size:7.5px;display:block;margin-top:1px">${r.ref}</small>`:"")+

   (r.notes?`<br><small style="color:#94a3b8;font-size:7.5px">${escapeHTML(r.notes)}</small>`:"");

  tableRows+=`<tr class="${r.opening?'opening':''}">

   <td class="col-seq">${r.seq}</td>

   <td class="col-date">${r.date}</td>

   <td class="col-desc">${descHTML}</td>

   <td class="col-amt num" style="color:${r.madden>0?'#991b1b':'#cbd5e1'}">${r.madden>0?money(r.madden):"—"}</td>

   <td class="col-amt num" style="color:${r.daen>0?'#047857':'#cbd5e1'}">${r.daen>0?money(r.daen):"—"}</td>

   <td class="col-amt num" style="color:${r.balance<0?'#047857':'#0f172a'}">${money(Math.abs(r.balance))}${r.balance<0?" له":""}</td>

  </tr>`;

 });

 const balanceBoxClass=isNegative?"negative":(isZero?"":"positive");

 const balanceLabel=isNegative?"الرصيد المستحق لصاحب الحساب":(isZero?"الحساب مسدَّد بالكامل":"الرصيد المستحق على صاحب الحساب");

  <div class="stmt-legal negative">

   <div class="title">⚖️ إقرار بحق مالي</div>

   <p>

    يُشهد بأن صاحب هذا الحساب قد أوفى بكامل ما في ذمته من التزامات مالية تجاه الطرف الثاني المذكور أعلاه، وزيادةً على ذلك بمبلغ <strong>${balanceDisplay}</strong>، وبذلك يُعدّ صاحب الحساب <strong>دائنًا</strong> بهذا المبلغ في ذمة الطرف الثاني.

    <br>

    ويُعتد بهذا الكشف كمستند رسمي يُثبت أحقية صاحب الحساب في المطالبة بالمبلغ المتبقي، ويُعمل به عند الحاجة أمام الجهات المختصة.

   </p>

  </div>`:`

  <div class="stmt-legal">

   <div class="title">📋 إقرار بالمبلغ المستحق</div>

   <p>

    يُشهد بأن صاحب هذا الحساب مدين للطرف الثاني المذكور أعلاه بمبلغ <strong>${balanceDisplay}</strong>، وذلك بموجب الحركات المالية المثبتة أعلاه.
